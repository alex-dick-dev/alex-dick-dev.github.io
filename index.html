// Typisierter gRPC Service Contract (.NET 10 / C# 14)
public sealed class OrderIntegrationService(
                        IOrderDispatcher dispatcher,
                        ILogger<OrderIntegrationService> logger)
    : Integration.IntegrationBase
{
                        public override async Task<SyncResponse> PublishBatch(
                        BatchRequest request,
                        ServerCallContext context)
    {
                        var processed = await dispatcher
            .DispatchAsync(request.Items, context.CancellationToken);

                        return new SyncResponse { ProcessedCount = processed };
    }
}
