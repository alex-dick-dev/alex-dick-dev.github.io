// High-Throughput gRPC Service Contract (.NET 8)
public sealed class SystemTelemetryService(
    IStateAggregator aggregator,
    ILogger<SystemTelemetryService> logger)
    : Telemetry.TelemetryBase
{
    public override async Task StreamDiagnostics(
        StreamRequest request,
        IServerStreamWriter<DiagnosticFrame> stream,
        ServerCallContext context)
    {
        await foreach (var frame in aggregator
            .ReadFramesAsync(request.NodeId, context.CancellationToken))
        {
            await stream.WriteAsync(frame);
        }
    }
}
